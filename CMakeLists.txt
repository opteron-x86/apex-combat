cmake_minimum_required(VERSION 3.28)
project(ApexCombat VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(APEX_BUILD_PLUGIN "Build the Windows SKSE plugin" ${WIN32})
option(APEX_BUILD_TESTS "Build portable combat tests" ON)
option(APEX_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)

add_library(apex_core STATIC src/core/CombatRules.cpp src/core/Configuration.cpp src/core/PoiseRules.cpp)
target_include_directories(apex_core PUBLIC src)
if(MSVC)
    target_compile_options(apex_core PRIVATE /EHsc /utf-8 /W4)
else()
    target_compile_options(apex_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(APEX_SANITIZERS)
        target_compile_options(apex_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(apex_core PUBLIC -fsanitize=address,undefined)
    endif()
endif()
if(APEX_BUILD_TESTS)
    enable_testing()
    add_executable(apex_core_tests tests/CoreTests.cpp)
    target_link_libraries(apex_core_tests PRIVATE apex_core)
    add_test(NAME combat_rules COMMAND apex_core_tests)
    add_executable(apex_configuration_tests tests/ConfigurationTests.cpp)
    target_link_libraries(apex_configuration_tests PRIVATE apex_core)
    add_test(NAME configuration COMMAND apex_configuration_tests ${CMAKE_CURRENT_SOURCE_DIR}/package/SKSE/Plugins/ApexCombat.ini)
    add_executable(apex_parry_contacts_tests tests/ParryContactsTests.cpp)
    target_link_libraries(apex_parry_contacts_tests PRIVATE apex_core)
    add_test(NAME parry_contacts COMMAND apex_parry_contacts_tests)
    add_executable(apex_poise_tests tests/PoiseTests.cpp)
    target_link_libraries(apex_poise_tests PRIVATE apex_core)
    add_test(NAME poise_rules COMMAND apex_poise_tests)
endif()

if(APEX_BUILD_PLUGIN)
    if(NOT WIN32)
        message(FATAL_ERROR "The SKSE plugin requires a Windows target toolchain.")
    endif()
    include(FetchContent)
    set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
    set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
    set(ENABLE_SKYRIM_VR ON CACHE BOOL "" FORCE)
    set(BUILD_TESTS OFF CACHE BOOL "CommonLib tests" FORCE)
    set(COMMONLIB_PREBUILT OFF CACHE BOOL "Build pinned CommonLib source" FORCE)
    set(COMMONLIB_PREBUILT_DIR "" CACHE PATH "" FORCE)
    set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
    set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(CommonLibSSE
        GIT_REPOSITORY https://github.com/alandtse/CommonLibSSE-NG.git
        GIT_TAG 12d61f3c39f7d4ba1b1ab543df2bca0e613dc747
        GIT_SUBMODULES extern/openvr
        EXCLUDE_FROM_ALL)
    FetchContent_MakeAvailable(CommonLibSSE)
    foreach(property COMPILE_DEFINITIONS INTERFACE_COMPILE_DEFINITIONS)
        get_target_property(definitions CommonLibSSE ${property})
        list(FILTER definitions EXCLUDE REGEX "^(WINVER|_WIN32_WINNT)=")
        list(APPEND definitions WINVER=0x0A00 _WIN32_WINNT=0x0A00)
        set_property(TARGET CommonLibSSE PROPERTY ${property} "${definitions}")
    endforeach()
    add_library(ApexCombat SHARED src/Plugin.cpp src/runtime/Runtime.cpp)
    target_link_libraries(ApexCombat PRIVATE apex_core CommonLibSSE)
    target_include_directories(ApexCombat PRIVATE src third_party)
    target_compile_definitions(ApexCombat PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE
        APEX_VERSION="${PROJECT_VERSION}"
        APEX_VERSION_MAJOR=${PROJECT_VERSION_MAJOR}
        APEX_VERSION_MINOR=${PROJECT_VERSION_MINOR}
        APEX_VERSION_PATCH=${PROJECT_VERSION_PATCH})
    target_compile_options(ApexCombat PRIVATE /EHsc /utf-8 /W4 /Zi)
    target_link_options(ApexCombat PRIVATE /DEBUG:FULL /Brepro /PDBALTPATH:ApexCombat.pdb)
    set_target_properties(ApexCombat PROPERTIES PREFIX "" OUTPUT_NAME ApexCombat)
    install(TARGETS ApexCombat RUNTIME DESTINATION SKSE/Plugins LIBRARY DESTINATION SKSE/Plugins)
    install(DIRECTORY package/ DESTINATION .)
endif()
